class StrategyBot:
    def __init__(self):
        self.rules = []

    def add_rule(self, rule_name, condition_fn, pass_msg, fail_msg):
        """Rule add karne ke liye"""
        self.rules.append({
            "name": rule_name,
            "check": condition_fn,
            "pass": pass_msg,
            "fail": fail_msg
        })

    def analyze(self, current_data):
        passed_reasons = []
        failed_reasons = []
        
        for rule in self.rules:
            if rule["check"](current_data):
                passed_reasons.append(rule["pass"])
            else:
                failed_reasons.append(rule["fail"])

        # Agar saare rules pass honge tabhi Entry allowed hogi
        can_enter = len(failed_reasons) == 0
        return can_enter, passed_reasons, failed_reasons

# --- BOT SETUP ---
bot = StrategyBot()

# Apne Rules yahan set karo (Example logic):
bot.add_rule(
    "RSI Check", 
    lambda d: d['rsi'] < 30, 
    "RSI 30 ke neeche hai (Oversold zone - Buying area)", 
    "RSI 30 se upar hai (Oversold nahi hua)"
)

bot.add_rule(
    "EMA Crossover", 
    lambda d: d['ema_fast'] > d['ema_slow'], 
    "Fast EMA, Slow EMA ke upar hai (Trend Bullish hai)", 
    "Fast EMA, Slow EMA ke neeche hai (Trend Clear nahi hai)"
)

bot.add_rule(
    "Volume Check", 
    lambda d: d['volume'] > d['avg_volume'], 
    "Volume average se ziada hai (Strong Breakout)", 
    "Volume kam hai (Fakeout ka risk hai)"
)

# --- USER QUERY FUNCTION ---
def ask_bot(market_data):
    can_enter, passed, failed = bot.analyze(market_data)
    
    print("\n-------------------------------------------")
    if can_enter:
        print("✅ DECISION: HAAN, ABI ENTRY LE SAKTE HO!")
        print("\n--- Kyun Leni Hai Entry? ---")
        for p in passed:
            print(f"• {p}")
    else:
        print("❌ DECISION: ABI ENTRY NAHI LENI HAI!")
        print("\n--- Kyun NAHI Leni Entry? (Wajah) ---")
        for f in failed:
            print(f"• {f}")
        
        if passed:
            print("\n--- Ye Conditions Sahi Thain: ---")
            for p in passed:
                print(f"• {p}")
    print("-------------------------------------------\n")
